    <div id="chart-container">
        <canvas id="stock-chart"></canvas>
    </div>

    <script>
        const csvUrl = 'https://raw.githubusercontent.com/shaktids/stock_app_test/refs/heads/main/dump.csv';
        let companyData = [];

        // Fetch and parse the CSV data
        fetch(csvUrl)
            .then(response => response.text())
            .then(data => {
                // Parse CSV using PapaParse
                Papa.parse(data, {
                    header: true,
                    skipEmptyLines: true,
                    complete: function(results) {
                        companyData = results.data;
                        populateCompanyList();
                    }
                });
            })
            .catch(error => console.error('Error fetching CSV:', error));

        // Populate the company list
        function populateCompanyList() {
            const companyNames = [...new Set(companyData.map(item => item.Company))]; // Get unique company names
            const companyListElement = document.getElementById('companies');
            companyNames.forEach(company => {
                const li = document.createElement('li');
                li.textContent = company;
                li.addEventListener('click', () => displayChartForCompany(company));
                companyListElement.appendChild(li);
            });
        }

        // Display chart for selected company
        function displayChartForCompany(company) {
            const filteredData = companyData.filter(item => item.Company === company);
            const labels = filteredData.map(item => item.Date);
            const stockPrices = filteredData.map(item => parseFloat(item.Close));

            const ctx = document.getElementById('stock-chart').getContext('2d');
            const chart = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: labels,
                    datasets: [{
                        label: ${company} Stock Price,
                        data: stockPrices,
                        borderColor: 'rgb(75, 192, 192)',
                        tension: 0.1
                    }]
                },
                options: {
                    responsive: true,
                    scales: {
                        x: {
                            type: 'category',
                            title: {
                                display: true,
                                text: 'Date'
                            }
                        },
                        y: {
                            title: {
                                display: true,
                                text: 'Stock Price ($)'
                            }
                        }
                    }
                }
            });
        }
    </script>
